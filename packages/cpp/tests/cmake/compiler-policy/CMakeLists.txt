cmake_minimum_required(VERSION 3.25)

project(afd_compiler_policy_test NONE)

if(NOT DEFINED AFD_SOURCE_DIR OR NOT DEFINED AFD_TEST_SCENARIO)
    message(FATAL_ERROR "AFD_SOURCE_DIR and AFD_TEST_SCENARIO are required")
endif()

include(${AFD_SOURCE_DIR}/cmake/AfdTargetOptions.cmake)

function(assert_has options expected)
    list(FIND options "${expected}" index)
    if(index EQUAL -1)
        message(FATAL_ERROR "${AFD_TEST_SCENARIO}: expected '${expected}' in '${options}'")
    endif()
endfunction()

function(assert_lacks options unexpected)
    list(FIND options "${unexpected}" index)
    if(NOT index EQUAL -1)
        message(FATAL_ERROR "${AFD_TEST_SCENARIO}: did not expect '${unexpected}' in '${options}'")
    endif()
endfunction()

if(AFD_TEST_SCENARIO STREQUAL "msvc-supported")
    afd_require_supported_compiler("MSVC" "19.30")
    afd_compiler_options(options "MSVC" "19.30" TRUE TRUE)
    assert_has("${options}" "/W4")
    assert_has("${options}" "/WX")
    assert_lacks("${options}" "-Wno-missing-designated-field-initializers")
elseif(AFD_TEST_SCENARIO STREQUAL "msvc-too-old")
    afd_require_supported_compiler("MSVC" "19.29.30159")
elseif(AFD_TEST_SCENARIO STREQUAL "clang-cl")
    # clang-cl can use the Microsoft frontend and VS 2019 headers without being cl.exe 19.29.
    afd_require_supported_compiler("Clang" "23.1.2")
    afd_require_clang_cl_regression_toolchain("Clang" "23.1.2" "MSVC" "23.1.2")
    afd_compiler_options(options "Clang" "23.1.2" TRUE TRUE)
    assert_has("${options}" "/W4")
    assert_has("${options}" "/WX")
    assert_has("${options}" "-Wno-missing-field-initializers")
    assert_has("${options}" "-Wno-missing-designated-field-initializers")
    assert_lacks("${options}" "-Werror")
elseif(AFD_TEST_SCENARIO STREQUAL "clang-cl-too-old")
    afd_require_clang_cl_regression_toolchain("Clang" "20.1.8" "MSVC" "22.1.8")
elseif(AFD_TEST_SCENARIO STREQUAL "clang-cl-unexpected-version")
    afd_require_clang_cl_regression_toolchain("Clang" "22.1.8" "MSVC" "23.1.2")
elseif(AFD_TEST_SCENARIO STREQUAL "clang-gnu-frontend")
    afd_require_supported_compiler("Clang" "21.0")
    afd_compiler_options(options "Clang" "21.0" FALSE TRUE)
    assert_has("${options}" "-Wall")
    assert_has("${options}" "-Werror")
    assert_has("${options}" "-Wno-missing-designated-field-initializers")
    assert_lacks("${options}" "/WX")
elseif(AFD_TEST_SCENARIO STREQUAL "apple-clang-before-subgroup")
    afd_compiler_options(options "AppleClang" "17.0" FALSE TRUE)
    assert_has("${options}" "-Wno-missing-field-initializers")
    assert_lacks("${options}" "-Wno-missing-designated-field-initializers")
else()
    message(FATAL_ERROR "Unknown AFD_TEST_SCENARIO: ${AFD_TEST_SCENARIO}")
endif()
